find_package(Doxygen REQUIRED dot)
find_package(Sphinx REQUIRED)

# Keep generated documentation up to date when public headers or documentation
# sources change. CONFIGURE_DEPENDS also notices newly added files.
file(GLOB_RECURSE SIOXX_PUBLIC_HEADERS CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/include/sioxx/*.hpp"
)
file(GLOB_RECURSE SPHINX_SOURCE_FILES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/*.rst"
  "${CMAKE_CURRENT_SOURCE_DIR}/_diagrams/*"
  "${CMAKE_CURRENT_SOURCE_DIR}/_static/*"
  "${CMAKE_CURRENT_SOURCE_DIR}/_templates/*"
)

set(DOXYGEN_INPUT_DIR ${PROJECT_SOURCE_DIR}/include/sioxx)
set(DOXYGEN_OUTPUT_DIR ${CMAKE_CURRENT_BINARY_DIR}/doxygen)
set(DOXYGEN_INDEX_FILE ${DOXYGEN_OUTPUT_DIR}/html/index.html)
set(DOXYGEN_XML_INDEX_FILE ${DOXYGEN_OUTPUT_DIR}/xml/index.xml)
set(DOXYGEN_STAMP_FILE ${DOXYGEN_OUTPUT_DIR}/doxygen.stamp)
set(DOXYFILE_IN ${CMAKE_CURRENT_SOURCE_DIR}/Doxyfile.in)
set(DOXYFILE_OUT ${CMAKE_CURRENT_BINARY_DIR}/Doxyfile)

configure_file(${DOXYFILE_IN} ${DOXYFILE_OUT} @ONLY)

add_custom_command(
  OUTPUT ${DOXYGEN_STAMP_FILE}
  BYPRODUCTS
    ${DOXYGEN_INDEX_FILE}
    ${DOXYGEN_XML_INDEX_FILE}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${DOXYGEN_OUTPUT_DIR}
  COMMAND ${DOXYGEN_EXECUTABLE} ${DOXYFILE_OUT}
  COMMAND ${CMAKE_COMMAND} -E touch ${DOXYGEN_STAMP_FILE}
  DEPENDS
    ${SIOXX_PUBLIC_HEADERS}
    ${DOXYFILE_IN}
  MAIN_DEPENDENCY ${DOXYFILE_OUT}
  COMMENT "Generating API documentation with Doxygen"
  VERBATIM
)

add_custom_target(Doxygen ALL DEPENDS ${DOXYGEN_STAMP_FILE})

set(SPHINX_SOURCE ${CMAKE_CURRENT_SOURCE_DIR})
set(SPHINX_BUILD ${CMAKE_CURRENT_BINARY_DIR}/sphinx)
set(SPHINX_INDEX_FILE ${SPHINX_BUILD}/index.html)
set(SPHINX_STAMP_FILE ${SPHINX_BUILD}/sphinx.stamp)

add_custom_command(
  OUTPUT ${SPHINX_STAMP_FILE}
  BYPRODUCTS ${SPHINX_INDEX_FILE}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${SPHINX_BUILD}
  COMMAND
    ${SPHINX_EXECUTABLE}
    -W
    --keep-going
    -b html
    -Dbreathe_projects.sioxx=${DOXYGEN_OUTPUT_DIR}/xml
    -Dgraphviz_dot=${DOXYGEN_DOT_EXECUTABLE}
    ${SPHINX_SOURCE}
    ${SPHINX_BUILD}
  COMMAND ${CMAKE_COMMAND} -E touch ${SPHINX_STAMP_FILE}
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  DEPENDS
    ${SPHINX_SOURCE_FILES}
    ${DOXYGEN_STAMP_FILE}
  MAIN_DEPENDENCY ${SPHINX_SOURCE}/conf.py
  COMMENT "Generating HTML documentation with Sphinx"
  VERBATIM
)

add_custom_target(Sphinx ALL DEPENDS ${SPHINX_STAMP_FILE})
add_dependencies(Sphinx Doxygen)

install(DIRECTORY ${SPHINX_BUILD}/ DESTINATION ${CMAKE_INSTALL_DOCDIR})
